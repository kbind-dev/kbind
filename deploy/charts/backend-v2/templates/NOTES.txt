Curate the catalog by labeling exported CRDs and creating Exports:

  kubectl label crd mangodbs.mangodb.io "core.kbind.io/exported=true"
  kubectl apply -f - <<'EOF'
  apiVersion: catalog.kbind.io/v1alpha1
  kind: Export
  metadata:
    name: mangodb
  spec:
    title: MangoDB
    apis:
      - name: mangodbs.mangodb.io
  EOF
